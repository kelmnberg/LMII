int main() {

    ptr = (int *)malloc(sizeof(int));
    *ptr = 5;
    if (ptr == NULL) {
        printf("Memory allocation failed\n");
        return 1;}

    printf("La valeur pointée par ptr est : %d\n", *ptr);

    free(ptr);
    return 0;
}

#include <stdio.h>

int Max(int *t, int size) {
    int max = *t;
    for (int i = 1; i < size; i++) {
        if (*(t + i) > max) {
            max = *(t + i);
        }
    }
    return max;
}

int main() {
    int t[100];
    int size;

    printf("donner le taille: ");
    scanf("%d", &size);

    printf("donner les elements:\n");
    for (int i = 0; i < size; i++) {
        scanf("%d", &t[i]);
    }

    int max = Max(t, size);
    printf("le max est %d\n", max);

    return 0;
}
